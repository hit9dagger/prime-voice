cmake_minimum_required(VERSION 3.22)
project(primevoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  add_library(primevoice SHARED bridge.cpp)
  target_compile_definitions(primevoice PRIVATE PRIME_AI_UNSUPPORTED=1)
  return()
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(NOT PRIME_LLAMA_SOURCE)
  include(FetchContent)
  FetchContent_Declare(llama URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/2d8a32502909ca065c41c4cf5f2fb1c427c047fc
    URL_HASH SHA256=7befec5f2a7fbd7fc69b69e4f80c9824a64cb760cd1b18aa0e1a6b1af53de7d8)
  FetchContent_MakeAvailable(llama)
else()
  add_subdirectory(${PRIME_LLAMA_SOURCE} llama)
endif()
if(ANDROID)
  add_library(primevoice SHARED bridge.cpp)
  target_link_libraries(primevoice PRIVATE llama android log)
  target_link_options(primevoice PRIVATE -Wl,-z,max-page-size=16384)
else()
  add_executable(prime-benchmark ../../../../tools/benchmark/runner.cpp)
  target_include_directories(prime-benchmark PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(prime-benchmark PRIVATE llama)
endif()
